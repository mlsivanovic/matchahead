import {createRoot} from 'react-dom/client';

const teams=selectableTeams();

return teams.map(team=>({teamId:team.id,sport:team.sport,seasonId:'2026-2027',kind:'verified-schedule',checkedAt:new Date(NOW).toISOString(),storedAt:new Date(NOW).toISOString(),response:{kind:'verified-schedule',teams,competitions:comps,changes:[],manifests:[],result:{teamId:team.id,sport:team.sport,seasonId:'2026-2027',checkedAt:new Date(NOW).toISOString(),futureFixtures:all.filter(f=>f.homeTeamId===team.id||f.awayTeamId===team.id),coverage:comps.map(c=>({id:c.id,competitionId:c.id,provider:'review-mock',publication:'allowed',scheduleAvailability:'published'}))}}}));}

useEffect(()=>{(window as any).reviewCount=setCount;(window as any).reviewTheme=(t:string)=>{setTheme(t);applyDocumentTheme(document,t as any,false)};(window as any).reviewOnline=setOnline;(window as any).reviewTab=setTab},[]);
const toggle=(id:string,values:string[],set:any)=>set(values.includes(id)?values.filter(x=>x!==id):[...values,id]);
const finder:any={sport:saved.sport,teamId,teams,seasonId:'2026-2027',working:false,error:null,lastGood:saved,displayed:saved,cooldown:{allowed:true,waitMs:0},snapshots:data,configured:true,focusTeam:(_sport:string,id:string)=>setTeamId(id),find:async()=>{},abortPending:()=>{}};
const account:any={status:'signed-in',uid:'review-only',email:'mls.ivanovic@gmail.com',profile:{timeZone:'Europe/Belgrade',reminderMinutes:30,notifyScheduleChange:true,notifyCancellation:true,favoriteTeamIds:[]},followedTeamIds:followed,manualFixtureIds:manual,message:null};
return <div className="app"><header className="top"><p className="brand">MatchAhead</p></header><main id="sadrzaj"><div hidden={tab!=='matches'}><PersonalAgenda active={tab==='matches'} now={NOW} timeZone="Europe/Belgrade" online={online} followed={followed} manualFixtureIds={manual} onToggleManual={id=>toggle(id,manual,setManual)} draftNote={note} onDraft={setNote} serverSnapshots={data as any} accountIdentity="signed-in:review-only"/></div><div hidden={tab!=='clubs'}><ClubsScreen active={tab==='clubs'} followed={followed} manualFixtureIds={manual} onToggle={id=>toggle(id,followed,setFollowed)} onToggleManual={id=>toggle(id,manual,setManual)} finder={finder} timeZone="Europe/Belgrade" now={NOW} online={online}/></div><div hidden={tab!=='settings'}><SettingsScreen active={tab==='settings'} prefs={{...prefs,theme}} onPrefs={setPrefs} onTheme={t=>{setTheme(t);applyDocumentTheme(document,t,false)}} onClear={()=>setNote('')} account={account} onSignOut={()=>{}} onSaveAccount={()=>{}} onDeleteAccount={()=>{}} install={{standalone:false,ios:false,canPrompt:false,onInstall:()=>{}}}/></div></main><nav className="nav" aria-label="Glavna navigacija">{[['matches','Utakmice'],['clubs','Klubovi'],['settings','Podešavanja']].map(([id,label])=><a key={id} href={`#${id}`} aria-current={tab===id?'page':undefined} onClick={()=>setTab(id)}>{label}</a>)}</nav></div>}
